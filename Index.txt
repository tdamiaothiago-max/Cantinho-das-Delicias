    // Load state from localStorage or initialize defaults
    let appData = JSON.parse(localStorage.getItem('cantinho_data')) || defaultData;
    let activeCategory = "Todas";

    function saveData() {
      localStorage.setItem('cantinho_data', JSON.stringify(appData));
      renderPage();
    }

    function renderPage() {
      // Set Year
      document.getElementById('year').innerText = new Date().getFullYear();

      // Info binding
      document.getElementById('info-address').innerText = appData.address;
      document.getElementById('info-phone').innerText = appData.phone;
      document.getElementById('info-hours').innerText = appData.hours;

      // WhatsApp links
      const waUrl = `https://wa.me/${appData.whatsapp}?text=Olá!%20Vim%20pelo%20site%20e%20gostaria%20de%20fazer%20um%20pedido.`;
      document.getElementById('nav-btn-whatsapp').href = waUrl;
      document.getElementById('hero-btn-pedir').href = waUrl;
      document.getElementById('btn-whatsapp-footer').href = waUrl;
      document.getElementById('btn-whatsapp-float').href = waUrl;

      // Status badge
      const statusBadge = document.getElementById('badge-status-nav');
      if (appData.status === "Aberto") {
        statusBadge.className = "inline-flex items-center px-2.5 py-1 rounded-full text-xs font-semibold bg-emerald-100 text-emerald-800";
        statusBadge.innerHTML = `<span class="w-2 h-2 mr-1.5 bg-emerald-500 rounded-full animate-pulse"></span> Aberto Agora`;
      } else {
        statusBadge.className = "inline-flex items-center px-2.5 py-1 rounded-full text-xs font-semibold bg-red-100 text-red-800";
        statusBadge.innerHTML = `<span class="w-2 h-2 mr-1.5 bg-red-500 rounded-full"></span> Fechado`;
      }

      // Render Category Tabs
      const categories = ["Todas", ...new Set(appData.menu.map(item => item.category))];
      const tabsContainer = document.getElementById('category-tabs');
      tabsContainer.innerHTML = categories.map(cat => `
        <button onclick="setCategory('${cat}')" class="px-4 py-2 rounded-full text-xs font-semibold transition ${cat === activeCategory ? 'bg-brand-orange text-white shadow-md' : 'bg-white text-gray-600 hover:bg-gray-100'}">
          ${cat}
        </button>
      `).join('');

      // Render Menu Items Grid
      const filteredMenu = activeCategory === "Todas" ? appData.menu : appData.menu.filter(i => i.category === activeCategory);
      const menuGrid = document.getElementById('menu-grid');
      menuGrid.innerHTML = filteredMenu.map(item => `
        <div class="bg-white rounded-2xl shadow-sm hover:shadow-md transition overflow-hidden border border-gray-100 flex flex-col">
          <img src="${item.img}" alt="${item.name}" class="h-48 w-full object-cover">
          <div class="p-5 flex-1 flex flex-col justify-between">
            <div>
              <div class="flex justify-between items-start mb-2">
                <h3 class="font-bold text-lg text-brand-dark">${item.name}</h3>
                <span class="text-xs bg-brand-orange/10 text-brand-orange font-bold px-2 py-1 rounded">R$ ${item.price}</span>
              </div>
              <p class="text-xs text-gray-500 mb-4">${item.desc}</p>
            </div>
            <a href="https://wa.me/${appData.whatsapp}?text=Olá!%20Gostaria%20de%20pedir:%20${encodeURIComponent(item.name)}" target="_blank" class="w-full bg-gray-50 hover:bg-brand-orange hover:text-white text-gray-700 font-semibold py-2 rounded-lg text-xs transition flex items-center justify-center gap-2">
              <i class="fa-brands fa-whatsapp"></i> Pedir este Prato
            </a>
          </div>
        </div>
      `).join('');

      // Render Reviews
      const reviewsGrid = document.getElementById('reviews-grid');
      reviewsGrid.innerHTML = appData.reviews.map(rev => `
        <div class="bg-white p-6 rounded-2xl shadow-sm border border-gray-100">
          <div class="flex text-yellow-400 text-xs mb-3">
            ${'<i class="fa-solid fa-star"></i>'.repeat(rev.stars)}
          </div>
          <p class="text-xs text-gray-600 italic mb-4">"${rev.text}"</p>
          <div class="text-xs font-bold text-gray-800">${rev.author}</div>
        </div>
      `).join('');

    function setCategory(cat) {
      activeCategory = cat;
      renderPage();
    }

    // ==================== ADMIN LOGIC ====================
    let isAdminLoggedIn = false;

    // Keyboard shortcut to open admin: Ctrl + Shift + A
    document.addEventListener('keydown', (e) => {
      if (e.ctrlKey && e.shiftKey && e.key.toLowerCase() === 'a') {
        e.preventDefault();
        toggleAdminModal();
      }
    });

    function toggleAdminModal() {
      const modal = document.getElementById('admin-modal');
      modal.classList.toggle('hidden');

      if (!modal.classList.contains('hidden') && isAdminLoggedIn) {
        populateAdminFields();
      }
    }

    function checkAdminPin() {
      const pin = document.getElementById('admin-pin').value;
      if (pin === "1234") {
        isAdminLoggedIn = true;
        document.getElementById('admin-login-view').classList.add('hidden');
        document.getElementById('admin-content-view').classList.remove('hidden');
        document.getElementById('admin-footer-actions').classList.remove('hidden');
        populateAdminFields();
      } else {
        alert("PIN Incorreto! Tente novamente (PIN Padrão: 1234).");
      }
    }

    function switchAdminTab(tab) {
      document.getElementById('tab-btn-general').className = tab === 'general' ? 'pb-2 border-b-2 border-brand-orange text-brand-orange' : 'pb-2 border-b-2 border-transparent text-gray-500 hover:text-gray-700';
      document.getElementById('tab-btn-menu').className = tab === 'menu' ? 'pb-2 border-b-2 border-brand-orange text-brand-orange' : 'pb-2 border-b-2 border-transparent text-gray-500 hover:text-gray-700';

      document.getElementById('admin-sec-general').classList.toggle('hidden', tab !== 'general');
      document.getElementById('admin-sec-menu').classList.toggle('hidden', tab !== 'menu');
    }

    function populateAdminFields() {
      document.getElementById('edit-phone').value = appData.phone;
      document.getElementById('edit-whatsapp').value = appData.whatsapp;
      document.getElementById('edit-address').value = appData.address;
      document.getElementById('edit-hours').value = appData.hours;
      document.getElementById('edit-status').value = appData.status;

      renderAdminMenuTable();
    }

    function renderAdminMenuTable() {
      const tbody = document.getElementById('admin-menu-table-body');
      tbody.innerHTML = appData.menu.map(item => `
        <tr>
          <td class="p-3 font-medium">${item.name}</td>
          <td class="p-3 text-gray-500">${item.category}</td>
          <td class="p-3">R$ ${item.price}</td>
          <td class="p-3 text-right">
            <button onclick="deleteMenuItem(${item.id})" class="text-red-500 hover:text-red-700 text-xs font-bold">
              <i class="fa-solid fa-trash"></i> Excluir
            </button>
          </td>
        </tr>
      `).join('');
    }

    function openAddMenuItemModal() {
      const name = prompt("Nome do prato:");
      if (!name) return;
      const category = prompt("Categoria (ex: Brasileira, Oriental, Italiana, Bebidas):", "Gerais");
      const price = prompt("Preço (ex: 29,90):", "25,00");
      const desc = prompt("Descrição curta:", "Prato saboroso feito na hora.");
      const img = prompt("URL da Imagem:", "https://images.unsplash.com/photo-1546069901-ba9599a7e63c?auto=format&fit=crop&w=500&q=80");

      const newItem = {
        id: Date.now(),
        name,
        category: category || "Gerais",
        price: price || "0,00",
        desc: desc || "",
        img: img || "https://images.unsplash.com/photo-1546069901-ba9599a7e63c?auto=format&fit=crop&w=500&q=80"
      };

      appData.menu.push(newItem);
      renderAdminMenuTable();
    }

    function deleteMenuItem(id) {
      if (confirm("Tem certeza que deseja remover este item?")) {
        appData.menu = appData.menu.filter(item => item.id !== id);
        renderAdminMenuTable();
      }
    }

    function saveAdminChanges() {
      appData.phone = document.getElementById('edit-phone').value;
      appData.whatsapp = document.getElementById('edit-whatsapp').value;
      appData.address = document.getElementById('edit-address').value;
      appData.hours = document.getElementById('edit-hours').value;
      appData.status = document.getElementById('edit-status').value;

      saveData();
      toggleAdminModal();
      alert("Alterações salvas com sucesso!");
    }

    function resetDefaultData() {
      if (confirm("Deseja restaurar as informações originais do Cantinho das Delícias?")) {
        appData = JSON.parse(JSON.stringify(defaultData));
        saveData();
        populateAdminFields();
        alert("Dados restaurados para os originais!");
      }
    }

    // Initial Render
    renderPage();
  </script>
